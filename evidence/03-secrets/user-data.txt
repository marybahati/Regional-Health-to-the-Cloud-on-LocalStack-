#!/bin/bash
# Receives the secret ARN and DB endpoint. NEVER the password.
# On LocalStack this file is world-readable under /var/lib/cloud/instances/<id>/.
set -euxo pipefail

mkdir -p /etc/app
cat > /etc/app/runtime.env <<EOF
SERVICE_NAME=service-b
DB_SECRET_ARN=arn:aws:secretsmanager:us-east-1:000000000000:secret:regional-health/service-b/db-VzOfhH
DB_HOST=mysql-1a2eab65-warga-lab.d.aivencloud.com
DB_PORT=18736
PORT=3000
BIND_HOST=127.0.0.1
AWS_ENDPOINT_URL=http://localhost.localstack.cloud:4566
AWS_ACCESS_KEY_ID=test
AWS_SECRET_ACCESS_KEY=test
AWS_DEFAULT_REGION=us-east-1
EOF
chmod 0644 /etc/app/runtime.env

# If the image entrypoint is waiting on this file, it will start nginx + node.
touch /etc/app/runtime.env.ready
echo "user-data wrote secret ARN arn:aws:secretsmanager:us-east-1:000000000000:secret:regional-health/service-b/db-VzOfhH (value never present)"
